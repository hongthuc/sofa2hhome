    // Lọc sản phẩm theo danh mục
    function filterCategory(categoryName) {
      const heading = document.getElementById('activeCategoryHeading');
      const grid = document.getElementById('mainProductGrid');
      const buttons = document.querySelectorAll('.cat-tab-btn');

      buttons.forEach(btn => {
        if (btn.getAttribute('data-cat') === categoryName) {
          btn.className = 'cat-tab-btn px-2.5 py-1 rounded text-xs font-bold bg-[#1b3b2b] text-white transition whitespace-nowrap';
        } else {
          btn.className = 'cat-tab-btn px-2.5 py-1 rounded text-xs font-semibold bg-gray-100 hover:bg-gray-200 text-gray-700 transition whitespace-nowrap';
        }
      });

      if (categoryName === 'all') {
        heading.textContent = 'NỘI THẤT PHÒNG KHÁCH - MẪU THỰC TẾ 2H-HOME';
        grid.innerHTML = productsData.map(createProductCardHTML).join('');
      } else {
        heading.textContent = `DANH MỤC: ${categoryName.toUpperCase()} - SOFA 2H-HOME`;
        const filtered = productsData.filter(p => p.category.toLowerCase().includes(categoryName.toLowerCase()));
        const listToRender = filtered.length > 0 ? filtered : productsData;
        grid.innerHTML = listToRender.map(createProductCardHTML).join('');
      }
    }

    // Tìm kiếm sản phẩm
    function handleSearch(keyword) {
      const cleanKey = keyword.trim().toLowerCase();
      const grid = document.getElementById('mainProductGrid');
      const heading = document.getElementById('activeCategoryHeading');
      if (!cleanKey) {
        filterCategory('all');
        return;
      }
      const matched = productsData.filter(p => 
        p.name.toLowerCase().includes(cleanKey) || p.category.toLowerCase().includes(cleanKey)
      );
      heading.textContent = `KẾT QUẢ TÌM KIẾM: "${keyword}" (${matched.length} MẪU)`;
      grid.innerHTML = (matched.length ? matched : productsData).map(createProductCardHTML).join('');
      document.getElementById('san-pham-all').scrollIntoView({ behavior: 'smooth' });
    }

    // Mở & Đóng Modal sản phẩm
    function openProductModal(id) {
      const product = productsData.find(p => p.id === id);
      if (!product) return;
      const modal = document.getElementById('productModal');
      const img = document.getElementById('modalImg');
      img.src = `https://lh3.googleusercontent.com/d/${product.driveId}`;
      img.onerror = function() {
        this.onerror = null;
        this.src = `https://drive.google.com/thumbnail?id=${product.driveId}&sz=w800`;
      };
      document.getElementById('modalSaleBadge').textContent = product.discount;
      document.getElementById('modalCat').textContent = product.category;
      document.getElementById('modalTitle').textContent = product.name;
      document.getElementById('modalOldPrice').textContent = product.oldPrice;
      document.getElementById('modalNewPrice').textContent = product.newPrice;
      document.getElementById('qNote').value = `Tôi quan tâm mẫu: ${product.name} (${product.newPrice})`;
      modal.classList.remove('hidden');
    }

    let currentSlide = 0;
    const slides = document.querySelectorAll('.banner-slide');
    const dots = document.querySelectorAll('.slide-dot');

    function goToSlide(index) {
      slides.forEach((slide, i) => {
        if (i === index) {
          slide.classList.remove('hidden', 'opacity-0');
          slide.classList.add('opacity-100');
        } else {
          slide.classList.add('hidden', 'opacity-0');
          slide.classList.remove('opacity-100');
        }
      });
      dots.forEach((dot, i) => {
        dot.className = i === index 
          ? 'slide-dot w-2.5 h-2.5 rounded-full bg-amber-400 transition' 
          : 'slide-dot w-2.5 h-2.5 rounded-full bg-white/50 transition';
      });
      currentSlide = index;
    }

    function nextSlide() {
      const next = (currentSlide + 1) % slides.length;
      goToSlide(next);
    }

    function prevSlide() {
      const prev = (currentSlide - 1 + slides.length) % slides.length;
      goToSlide(prev);
    }

    // Đồng hồ đếm ngược Flash Sale
    function startCountdown() {
      let totalSeconds = 2 * 86400 + 18 * 3600 + 45 * 60 + 30;
      setInterval(() => {
        if (totalSeconds <= 0) totalSeconds = 86400;
        totalSeconds--;
        const d = Math.floor(totalSeconds / 86400);
        const h = Math.floor((totalSeconds % 86400) / 3600);
        const m = Math.floor((totalSeconds % 3600) / 60);
        const s = totalSeconds % 60;
        document.getElementById('cdDays').textContent = String(d).padStart(2, '0');
        document.getElementById('cdHours').textContent = String(h).padStart(2, '0');
        document.getElementById('cdMins').textContent = String(m).padStart(2, '0');
        document.getElementById('cdSecs').textContent = String(s).padStart(2, '0');
      }, 1000);
    }
